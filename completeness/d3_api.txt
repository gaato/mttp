// usage
async fn client_usage(backend : &@mttp.Backend, clock : &@clock.Clock) -> Unit raise {
  let user : Json = @mttp.basic_request()
    .get("https://api.example.com/users/1")
    .response(@mttp.as_json().or_fail())
    .send(backend)
    .body
  let client = @mttp.Client::new(
    @mttp.FollowRedirects::new(backend),
    clock,
    retry=@mttp.RetryPolicy::conservative(),
  )
  // GitHub's GraphQL `accept` hook, now a reader: raising here is retried.
  let reader = @mttp.as_json().or_fail().map(j => {
    if j is { "errors": [{ "type": "RATE_LIMITED", .. }, ..], .. } {
      raise @mttp.RateLimited(retry_after_ms=None, headers=@model.Headers::new(), body=b"")
    }
    j
  })
  let r = client.send(@mttp.quick_request().post("https://api.github.com/graphql").bucket("graphql").response(reader))
  ignore((user, r))
}

// usage (from shape/src/example, checked; runtime-tested in shape-rt)
pub(all) struct User {
  id : Int
  name : String
} derive(ToJson, FromJson, Eq, Debug)
pub(all) struct GetUser {
  id : Int
  verbose : String?
} derive(Eq, Debug)
pub(all) suberror UserError {
  NotFound(Int)
} derive(ToJson, FromJson, Debug)

pub let get_user : @endpoint.Endpoint[Unit, GetUser, UserError, User] = @endpoint.Endpoint::new(
  Get,
  "/users/{id}",
  security=@endpoint.unit(),
  input=@endpoint.Input::map2(
    @endpoint.path("id", @endpoint.int().validate(
      id => if id > 0 { None } else { Some("must be positive") },
      schema_patch={ "minimum": 1 },
    )),
    @endpoint.query("verbose", @endpoint.string()).optional(),
    (id, verbose) => { id, verbose },
    g => (g.id, g.verbose),
  ),
  errors=@endpoint.ErrorOut::new(e => match e {
    UserError::_ as u => Some(u)
    _ => None
  }).variant(404, @endpoint.json(schema={ "type": "object" })),
  output=@endpoint.json_out(@endpoint.json(schema={ "type": "object" })),
  summary="Get a user",
  tags=["users"],
)

/// Discord's interactions route: signature headers are security inputs,
/// so a request without them is answered 401 before its body is read.
pub let interactions : @endpoint.Endpoint[(String, String, Bytes), Unit, Unit, Unit] = @endpoint.Endpoint::new(
  Post,
  "/",
  security=@endpoint.Input::map2(
    @endpoint.Input::map2(
      @endpoint.header("x-signature-ed25519", @endpoint.string()),
      @endpoint.header("x-signature-timestamp", @endpoint.string()),
      (s, t) => (s, t),
      p => p,
    ),
    @endpoint.body_bytes(),
    (st, body) => (st.0, st.1, body),
    t => ((t.0, t.1), t.2),
  ),
  input=@endpoint.unit(),
  errors=@endpoint.ErrorOut::none(),
  output=@endpoint.no_content(),
)

fn app() -> @eps.Router raise @eps.RouterError {
  @eps.Router::new([
    @eps.serve_public(get_user, async fn(input) {
      if input.id == 1 { { id: 1, name: "gaato" } } else { raise NotFound(input.id) }
    }),
  ])
}

async fn typed(client : @mttp.Client) -> Unit raise {
  let req = @epc.to_req(get_user, base="https://api.example.com", (), { id: 42, verbose: None })
  match client.send(req).body {
    Ok(user) => println(user.name)
    Err(Declared(NotFound(id))) => println("no user \{id}")
    Err(_) => println("unexpected")
  }
}
