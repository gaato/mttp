pub(all) suberror UserError {
  NotFound(Int)
  Forbidden
} derive(ToJson, FromJson, Debug)

let user_errors : @api.Errors[UserError] = @api.errors(
  e => match e {
    UserError::_ as u => Some(u)
    _ => None
  },
  status=e => match e {
    NotFound(_) => 404
    Forbidden => 403
  },
  codec=@api.json(name="UserError", schema={ "type": "array" }),
  statuses=[404, 403],
)

// shared package: no net import, so it builds for wasm-gc, js and the server alike
pub let get_user : @api.Endpoint[Unit, (Int, Int?), UserError, User] = @api.endpoint(
  Get,
  "/users/{id}",
  input=@api.map2(@api.path("id", @api.int), @api.query("v", @api.int), (id, v) => (id, v), t => t),
  output=@api.json_out(user),
  errors=user_errors,
  summary="Get a user",
)

pub let create_user : @api.Endpoint[String, NewUser, UserError, User] = @api.endpoint(
  Post,
  "/users",
  input=@api.json_body(@api.json(schema={ "type": "object" })),
  output=@api.json_out(user, status=201),
  errors=user_errors,
).secure(@api.bearer())

pub let ticks : @api.Endpoint[Unit, Unit, @api.NoError, @api.Events[Int]] = @api.endpoint(
  Get,
  "/ticks",
  input=@api.none(),
  output=@api.sse(@api.int),
  errors=@api.no_errors(),
)

fn app() -> @api.Router raise @api.RouteConflict {
  @api.Router::new([
    get_user.serve(async fn(input) {
      let (id, _) = input
      if id == 1 { { id: 1, name: "gaato" } } else { raise NotFound(id) }
    }),
    create_user.serve_as(
      async fn(token) { if token == "secret" { "admin" } else { raise Forbidden } },
      async fn(_who, body) { { id: 2, name: body.name } },
    ),
    ticks.serve(async fn(_) {
      @api.Events::new(async fn(emit) { for i in 0..<3 { emit(i) } })
    }),
  ])
}

test "one definition, both sides, in process" {
  @testing.run(async fn() {
    let svc = app().service()                        // a Router is a Service
    let u = get_user.call(svc, (1, None))            // typed client
    let r = get_user.call(svc, (42, Some(3))) catch {
      UserError::NotFound(id) => "not found \{id}"   // the handler's error, re-raised
      e => "other \{e}"
    } noraise {
      _ => "ok"
    }
    let made = create_user.call_as(svc, "secret", { name: "neko" })
    let got = []
    ticks.call(svc, ()).each(async fn(i) { got.push(i) })  // [0, 1, 2]
    let doc = @api.openapi(title="demo", version="1", [get_user.op(), create_user.op()])
    ignore((u, r, made, doc))
  })
}

async fn plain(token : String) -> User raise {
  let clock = @net.RealClock::new()
  let svc = @net.client()
    |> @client.redirects()
    |> @client.cookies(@cookie.CookieJar::new(clock))
    |> @client.defaults(base_url="https://api.example.com")
    |> @client.auth(Bearer(token))
    |> @client.retry(clock, total_ms=60_000)
  let id = 42
  @client.send(svc, @mttp.Request::get(@mttp.url(u => u <+ "/users/\{id}")), @client.json)
}
