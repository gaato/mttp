pub(all) struct Response[T] { body : T; code : @model.StatusCode; status_text : String; headers : @model.Headers; history : Array[@model.ResponseMetadata]; request : @model.RequestMetadata }
pub async fn[T] Request::send(self : Request[T], backend : &Backend) -> Response[T] {
  let raw = backend.send(self.raw)
  defer raw.body.close()                           // always consumed and closed
  let body = (self.response.read)(raw.meta, raw.body, self.raw.options)
  { body, code: raw.meta.code, status_text: raw.meta.status_text, headers: raw.meta.headers, history: raw.history, request: raw.request }
}

pub let get_user : @endpoint.Endpoint[String, (Int, String?), ApiError, User] = secured
  .get()
  .path("/users/{id}")
  .in_(@endpoint.in2(@endpoint.path_param("id", @endpoint.int_codec()), @endpoint.query("fields", @endpoint.string_codec).optional()))
  .out(@endpoint.json_body(user_schema))
  .summary("Get a user")
  .tag("users")

pub fn app() -> @server_interp.Router raise {                 // server
  let authenticate = async fn(token : String) -> Result[String, ApiError] {
    if token == "good" { Ok("alice") } else { Err(Unauthorized({ message: "bad token" })) }
  }
  @server_interp.Router::new([
    get_user.server_security_logic(authenticate).server_logic(async fn(_who, input) {
      let (id, _fields) = input
      if id == 1 { Ok({ id: 1, name: "gaato" }) } else { Err(NotFound({ message: "no user \{id}" })) }
    }),
  ])
}
pub async fn typed(backend : &@client.Backend, base : @model.Uri) -> String {   // client
  match @client_interp.to_secure_request(get_user, base, "token", (42, None)).send(backend).body {
    Value(Ok(user)) => user.name
    Value(Err(NotFound(b))) => "not found: " + b.message
    Value(Err(Unauthorized(_))) => "unauthorized"
    Failure(_) => "undecodable"
  }
}
pub fn docs() -> Json raise { @openapi.to_openapi([get_user.meta()], title="users", version="1").to_json() }
