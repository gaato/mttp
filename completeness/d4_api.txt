pub(all) suberror UserError { NotFound(Int); Forbidden }
let user_errors : @endpoint.Errors[UserError] = @endpoint.Errors::new(
  e => match e { UserError::_ as u => Some(u); _ => None },
  [ @endpoint.json_variant(404, @schema.jint.body()).contramap(u => match u { NotFound(id) => Some(id); _ => None }, id => NotFound(id)),
    @endpoint.empty_variant(403).contramap(u => match u { Forbidden => Some(()); _ => None }, _ => Forbidden) ])
pub let get_user : @endpoint.Endpoint[String, GetUser, UserError, User] = @endpoint.endpoint(Get, "/users/{id}")
  .security(@endpoint.bearer())
  .input(@endpoint.Input::map2(
    @endpoint.path("id", @schema.int.validate(@schema.min(1))),
    @endpoint.query("fields", @schema.string).optional(),
    (id, fields) => { id, fields }, g => (g.id, g.fields)))
  .errors(user_errors)
  .output(@endpoint.json_out(user_json.body()))
  .operation_id("getUser")
